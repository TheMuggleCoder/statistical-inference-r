# Question One (i)
mle.exp <- function(n,lambda){
  sample <- rexp(n,rate = lambda)
  lambda_sample <- n/sum(sample)
  return(list(sample=sample,lambda_sample=lambda_sample))
}

results <- numeric(1000)
# Create a for loop to loop the function over 1000 times
for (i in 1:1000) {
  result <- mle.exp(10,1)
  results[i] <- result$lambda_sample
}
results
# Calculate the mean and variance of the estimators
mean_10 <- base::mean(results)
variance_10 <- var(results)
bias_10 <- mean_10 - 1
bias_10squared <- bias_10^2
MSE_10 <- variance_10 + bias_10squared
