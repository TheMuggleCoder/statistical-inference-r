\section{Question One}
\textbf{Calculating Bias for an Exponential Distribution:}\\
\begin{equation}
    \hat{\lambda} = \frac{n}{\sum x_i}
    \label{1}
\end{equation}
\begin{equation}
    \operatorname{Bias}(\hat{\lambda}) = \text{E}[\hat{\lambda}] - \lambda
    \label{2}
\end{equation}
Substituting equation \eqref{1} in to the expected value, we get: \begin{align}
    \text{E}[\hat{\lambda}] &= \text{E}[\frac{n}{\sum x_i}]\\
    &= n\text{E}[\frac{1}{\sum x_i}]
    \label{3}
    \end{align}
If we let $\text{S} = \sum x_i$, we know that S has a Gamma distribution because we are letting S be the sum of n Exponential distributions.\\
So, from equation \eqref{3} we get: \begin{equation}
    \text{E}[\hat{\lambda}] = n\text{E}[\frac{1}{S}] 
    \label{5}
\end{equation}
where $S \sim Gamma(n,\lambda)$.\\
Using the Gamma Density function we can find the Expected value of the reciprocal of S: \begin{align}
    \text{E}[\frac{1}{S}] &= \int_0 ^\infty \frac{1}{s}\frac{\lambda^n}{\Gamma(n)} s^{n-1} e^{-\lambda s} ds\\
    &= \frac{\lambda^n}{\Gamma(n)}\int_0 ^{\infty} s^{n-2}e^{\lambda s}ds
\end{align}
Using the Gamma Integral Identity: $\int_0 ^\infty s^{\alpha-1}e^{-\lambda s}ds=\frac{\Gamma(\alpha)}{\lambda^\alpha}$ we get \begin{align}
    \text{E}[\frac{1}{S}] &= \frac{\lambda^n}{\Gamma(n)}\frac{\Gamma(n-1)}{\lambda^{n-1}}\\
    &= \frac{\lambda \Gamma(n-1)}{\Gamma(n)}
\end{align}
\newpage
Use the fact that $\Gamma(n) = (n-1)\Gamma(n-1)$. Therefore: \begin{align}
    \text{E}[\frac{1}{S}] &= \frac{\lambda \Gamma(n-1)}{(n-1)\Gamma(n-1)}\\
    &= \frac{\lambda}{n-1}
    \label{11}
\end{align}
So, from equation \eqref{5} we have: \begin{equation}
    \text{E}[\hat{\lambda}] = \frac{n\lambda}{n-1}
\end{equation}
Substituting this into equation \eqref{2} we get: \begin{align}
    \text{Bias}(\hat{\lambda}) &= \frac{n\lambda}{n-1} - \lambda \\
    &= \frac{\lambda}{n-1}
    \label{bias}
\end{align}
\textbf{Calculating Efficiency for an Exponential Distribution:}\\
\begin{equation}
    \text{eff}(\hat{\lambda}) = \frac{\text{CRLB}}{\text{Var}(\hat{\lambda})}
    \label{6}
\end{equation}
So, in order to calculate the efficiency, we need to calculate the Variance and the Cramer-Rao Lower Bounds separately first.\begin{equation}
    \text{CRLB} = \frac{1}{\mathcal{I}(\hat{\lambda})}
    \label{7}
\end{equation}
where $\mathcal{I}(\hat{\lambda})$ is the Expected information of the maximum likelihood estimator $\hat{\lambda}$.\begin{equation}
    \mathcal{I}(\lambda) = \text{E}\left[-\frac{\partial^2 \ell(\lambda)}{\partial \lambda^2}\right]
    \label{8}
\end{equation}
We know that \begin{equation}
    \frac{\partial\ell(\lambda)}{\partial \lambda} = \frac{n}{\lambda} - \sum x_i
\end{equation}
So, \begin{equation}
    \frac{\partial^2\ell(\lambda)}{\partial \lambda^2} = -\frac{n}{\lambda^2}
\end{equation}\begin{equation}
    \implies \mathcal{I}(\hat{\lambda}) = \frac{n}{\lambda^2}
\end{equation} \begin{equation}
   \implies \text{CRLB} = \frac{\lambda^2}{n}
\end{equation}
\newpage
\begin{equation}
   \text{Var}(\hat{\lambda}) = \text{Var}(\frac{n}{\sum x_i}) = n^2\text{Var}(\frac{1}{S})
\end{equation}
where, once again, $S \sim \text{Gamma}(n,\lambda)$. From the properties of Expectation and Variance we know that \begin{equation}
    \text{Var}(\frac{1}{S^2}) = \text{E}[\frac{1}{S^2}] - \big[\text{E}[\frac{1}{S}] \big]^2
    \label{var}
\end{equation}
We know from equation \eqref{11} that \begin{equation}
    \big[\text{E}[\frac{1}{S}] \big]^2=  \frac{\lambda^2}{(n-1)^2}
\end{equation}
So we need to calculate $\text{E}[\frac{1}{S^2}]$ using the Gamma density function once again. \begin{align}
    \text{E}[\frac{1}{S^2}] &= \int_0 ^\infty \frac{1}{s^2}\frac{\lambda^n}{\Gamma(n)}s^{n-1}e^{-\lambda s}ds\\
    &= \frac{\lambda^n}{\Gamma(n)}\int_0 ^\infty s^{n-3}e^{-\lambda s}ds
\end{align}
Using the Gamma Integral Identity function once again we get:\begin{align}
   \text{E}[\frac{1}{S^2}]&= \frac{\lambda^n}{\Gamma(n)}\frac{\Gamma(n-2)}{\lambda^{n-2}}\\
   &= \frac{\lambda^2 \Gamma(n-2)}{\Gamma(n)}
\end{align}
Since $\Gamma(n) = (n-1)\Gamma(n-1)$, we can say that $\Gamma(n-1) = (n-2)\Gamma(n-2)$, which means that $\Gamma(n) = (n-1)(n-2)\Gamma(n-2)$ \begin{align}
    \implies \text{E}[\frac{1}{S^2}] &= \frac{\lambda^2\Gamma(n-2)}{(n-1)(n-2)\Gamma(n-2)}\\
    &= \frac{\lambda^2}{(n-1)(n-2)}
\end{align}
Substituting into \eqref{var} we get: \begin{equation}
    \text{Var}(\frac{1}{S^2}) = \frac{\lambda^2}{(n-1)^2} - \frac{\lambda^2}{(n-1)(n-2)}
\end{equation}
which simplifies to:\begin{equation}
    \text{Var}(\frac{1}{S^2}) = \frac{\lambda^2}{(n-1)^2(n-2)}
\end{equation}
\begin{equation}
    \implies \text{Var}(\hat{\lambda} ) = \frac{n^2 \lambda^2}{(n-1)^2 (n-2)}
    \label{variance}
\end{equation}
\newpage
Therefore \begin{align}
    \text{eff}(\hat{\lambda}) &= \frac{\frac{\lambda^2}{n}}{\frac{n^2\lambda^2}{(n-1)^2(n-2)}}\\
    & = \frac{\frac{1}{n}}{\frac{n^2}{(n-1)^2(n-2)}} \\
    &= \frac{(n-1)^2(n-2)}{n^3}
\end{align}
\textbf{Note:} The $\lambda$'s cancel out and so changing the value of $\lambda$ will have no impact on the graphs.\\
\textbf{Calculating Mean Squared Error for an Exponential Distribution:}\\
\begin{equation}
    \text{MSE}(\hat{\lambda}) = \text{Var}(\hat{\lambda}) + \big[\text{Bias}(\lambda)\big]^2
\end{equation}
Using the results from equations \eqref{bias} and \eqref{variance} we get: \begin{equation}
   \text{MSE}(\hat{\lambda}) =\frac{n^2 \lambda^2}{(n-1)^2(n-2)} + \big[\frac{\lambda}{n-1}\big]^2
\end{equation}
Multiplying the right fraction by $(n-2)$ and simplifying down, we get: \begin{equation}
    \text{MSE}(\hat{\lambda}) = \frac{\lambda^2(n+2)}{(n-1)(n-2)}
\end{equation}
The sample size $(n)$ values of $(10,20,50,200,400,1000)$ and the $\lambda$ values of $(0.5,1,2)$ were substituted into the equations for the Bias, the efficiency and the mean squared error calculated above to produce all of the graphs below.\\

\section{Question Three}
\textbf{Computing the Maximum Likelihood Estimator for $\theta$:}

\begin{equation}
    f(x) = cx^{\theta-1}
    \label{density}
\end{equation}
Find the value of the normalising constant $c$, using a maximum x value of 12 computed in R:

\begin{equation}
    \int_0^{12} cx^{\theta-1}dx =1
    \label{density2}
\end{equation}
Left hand side:
\begin{align}
    \int_0^{12} cx^{\theta-1}dx 
    &= \left.\frac{cx^\theta}{\theta}\right|_0^{12}\\
    &= \frac{c12^\theta}{\theta}
\end{align}
From equation \eqref{density2}:

\begin{equation}
    \frac{c12^\theta}{\theta} = 1\\
    \implies c = \frac{\theta}{12^\theta}
\end{equation}
Substitute this value for $c$ into the log likelihood function $\ell(\theta)$.

\begin{align}
    \ell(\theta) &= \sum_{i=1}^n \log(f(x_i))\\
    &= \sum_{i=1}^n \log\left(\frac{\theta x_i^{\theta-1}}{12^\theta}\right)\\
    &= \sum_{i=1}^n \left[\log(\theta)+(\theta-1)\log(x_i)-\theta\log(12)\right]\\
    &= n\log(\theta)+(\theta-1)\sum_{i=1}^n\log(x_i)-n\theta\log(12)
\end{align}
To obtain the maximum likelihood estimator we must differentiate the log likelihood with respect to $\theta$.

\begin{align}
    \frac{\partial\ell(\theta)}{\partial\theta}
    &= \frac{n}{\theta}+\sum_{i=1}^n\log(x_i)-n\log(12)
    \label{diff}
\end{align}
Set equation \eqref{diff} to zero.

\begin{align}
    \frac{n}{\theta}+\sum_{i=1}^n\log(x_i)-n\log(12) &= 0\\
    \frac{n}{\theta} &= n\log(12)-\sum_{i=1}^n\log(x_i)\\
    \implies \hat{\theta}_{MLE} &= \frac{n}{n\log(12)-\sum_{i=1}^n\log(x_i)}
\end{align}

\noindent \textbf{Computing the Method of Moments Estimator for $\theta$:}\\
\begin{equation}
    \text{E}[X] = \bar{X}
    \label{moments}
\end{equation}
where $\bar{X}$ represents the sample mean.\\
Calculate the expected value of $X$: \begin{align}
    \text{E}[X] &= \int_0^{12} xf(x)dx \\
    &= \int_0^{12} x \frac{\theta}{12^\theta} x^{\theta -1}dx \\
    &= \int_0^{12}\frac{\theta}{12^\theta} x^\theta dx \\
    &= \left.\frac{\theta x^{\theta+1}}{12^\theta (\theta+1)}\right|_0^{12}\\
    &= \frac{12\theta}{\theta+1}
\end{align}
Substitute this value into equation \eqref{moments}:\begin{align}
    \frac{12\theta}{\theta+1} &= \bar{X}\\
    \implies \frac{\theta+1}{12\theta} = \frac{1}{\bar{X}}\\
    \implies \frac{1}{12}+\frac{1}{12\theta} = \frac{1}{\bar{X}}\\
    \implies \frac{1}{12\theta} = \frac{1}{\bar{X}}-\frac{1}{12} \\
    \implies \hat{\theta} = \frac{\bar{X}}{12-\bar{X}}
\end{align}
\newpage
\noindent \textbf{Deriving the Cumulative Distribution Function $F(x)$:}\\
\begin{align}F(x) 
&= \int_0^x f(u)du\\
&= \int_0^x \frac{\theta}{12^\theta} u^{\theta-1}du\\
&= \left.\frac{u^\theta}{12^\theta}\right|_0^x \\
&= \frac{x^\theta}{12^\theta}
\end{align}
\textbf{Deriving the Inverse Cumulative Distribution Function $F^{-1}(x)$:}\\
In order to find the inverse cumulative distribution function, we have to isolate $x$: \begin{align}
    \text{Let $y$} &= \Big(\frac{x}{12}\Big)^\theta
\end{align}
Take both sides to the power of $\frac{1}{\theta}$ to get: \begin{align}
    y^{\frac{1}\theta} &= \frac{x}{12} \\
    \implies x &= 12y^{\frac{1}{\theta}}
\end{align}
Therefore,\begin{equation}
    F^{-1}(x) = 12x^{\frac{1}{\theta}}
\end{equation}

\section{Question Four}
\textbf{Bootstrapping:} A bootstrap confidence interval is a range of values that is calculated without the need for assumptions on normal distributions or strict mathematical formulas. The idea of bootstrapping is to take a certain number of new samples, all of the same size from the original data, but there is replacement which means that some data points are kept and others are left out. The mean or median is calculated for each of the new samples and then a confidence interval is created based on this distribution.
